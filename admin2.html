/* ===================================================================== RUBRIQUE 8 (JS) : CONVERSION SÉCURISÉE DES LIENS GOOGLE DRIVE ===================================================================== */ function resoudrePhotoUrl(url) { if (!url || typeof url !== "string") return LOGO_FALLBACK_URL; const propre = url.trim(); const lower = propre.toLowerCase(); // Si la cellule est vide ou indique "attente" / "en attente", on attribue le logo ASTER if (!propre || lower === "attente" || lower === "en attente") { return LOGO_FALLBACK_URL; } if (propre.includes("lh3.googleusercontent.com/d/")) return propre; const matchSlash = propre.match(/\/d\/([a-zA-Z0-9_-]+)/); if (matchSlash && matchSlash[1]) { return "https://lh3.googleusercontent.com/d/" + matchSlash[1]; } const matchQuery = propre.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (matchQuery && matchQuery[1]) { return "https://lh3.googleusercontent.com/d/" + matchQuery[1]; } return propre; }
